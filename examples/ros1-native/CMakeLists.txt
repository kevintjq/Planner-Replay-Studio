cmake_minimum_required(VERSION 3.16)
project(planner_replay_bridge LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(catkin REQUIRED COMPONENTS roscpp topic_tools)
find_package(Qt5 REQUIRED COMPONENTS Core Network)
add_executable(planner_replay_bridge bridge.cpp)
target_include_directories(planner_replay_bridge PRIVATE ${catkin_INCLUDE_DIRS} ../../native/src)
target_link_libraries(planner_replay_bridge PRIVATE ${catkin_LIBRARIES} Qt5::Core Qt5::Network)

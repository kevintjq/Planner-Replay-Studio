if(WIN32)
    file(GLOB FFMPEG_RUNTIME_DLLS "${FFmpeg_ROOT}/bin/*.dll")
    if(NOT FFMPEG_RUNTIME_DLLS)
        message(FATAL_ERROR "FFmpeg_ROOT must include the shared runtime DLLs in bin")
    endif()
    # All executables (including tests) share this directory. Match the linked ABI.
    foreach(runtime IN LISTS FFMPEG_RUNTIME_DLLS)
        file(COPY "${runtime}" DESTINATION "${CMAKE_CURRENT_BINARY_DIR}/Release")
        file(COPY "${runtime}" DESTINATION "${CMAKE_CURRENT_BINARY_DIR}/Debug")
    endforeach()
    install(FILES ${FFMPEG_RUNTIME_DLLS} DESTINATION .)
endif()

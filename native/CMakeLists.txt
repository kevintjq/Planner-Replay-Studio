cmake_minimum_required(VERSION 3.22)
project(PlannerReplayNative VERSION 0.1.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)

find_package(Qt6 6.8 REQUIRED COMPONENTS Quick QuickControls2 OpenGL Concurrent Multimedia)
find_package(BZip2 REQUIRED)
find_package(yaml-cpp CONFIG REQUIRED)
find_package(lz4 CONFIG QUIET)
if(TARGET lz4::lz4)
    set(LZ4_LIBRARY lz4::lz4)
elseif(TARGET LZ4::lz4_shared)
    set(LZ4_LIBRARY LZ4::lz4_shared)
else()
    find_path(LZ4_INCLUDE_DIR NAMES lz4frame.h REQUIRED)
    find_library(LZ4_LIBRARY NAMES lz4 REQUIRED)
endif()

add_library(replay_data STATIC src/replay_data.cpp src/project_document.cpp src/video_timing.cpp src/rviz_layout.cpp)
target_include_directories(replay_data PUBLIC src)
if(LZ4_INCLUDE_DIR)
    target_include_directories(replay_data PRIVATE "${LZ4_INCLUDE_DIR}")
endif()
target_link_libraries(replay_data PUBLIC Qt6::Gui PRIVATE BZip2::BZip2 ${LZ4_LIBRARY} yaml-cpp::yaml-cpp)
qt_add_executable(PlannerReplayNative WIN32 src/main.cpp src/cloud_view.cpp src/cloud_view.h
    src/replay_controller.cpp src/replay_controller.h)
qt_add_resources(PlannerReplayNative "interface" PREFIX "/" FILES qml/Main.qml qml/FlightClip.qml qml/VideoTimeline.qml
    qml/SceneOverlays.qml qml/SceneLayerEditor.qml qml/MarkerTimeline.qml qml/LayoutEditor.qml)
target_link_libraries(PlannerReplayNative PRIVATE replay_data Qt6::Quick Qt6::QuickControls2 Qt6::OpenGL Qt6::Concurrent Qt6::Multimedia)
if(MSVC)
    target_compile_options(replay_data PRIVATE /utf-8)
    target_compile_options(PlannerReplayNative PRIVATE /utf-8)
endif()
install(TARGETS PlannerReplayNative RUNTIME DESTINATION .)

include(CTest)
if(BUILD_TESTING)
    add_executable(replay_data_test tests/replay_data_test.cpp)
    target_link_libraries(replay_data_test PRIVATE replay_data)
    add_test(NAME replay_data COMMAND replay_data_test)
    add_executable(project_document_test tests/project_document_test.cpp)
    target_link_libraries(project_document_test PRIVATE replay_data)
    add_test(NAME project_document COMMAND project_document_test)
    if(MSVC)
        target_compile_options(project_document_test PRIVATE /utf-8)
    endif()
    add_test(NAME app_smoke COMMAND PlannerReplayNative --smoke)
    set_tests_properties(app_smoke PROPERTIES TIMEOUT 20)
    add_test(NAME png_smoke COMMAND PlannerReplayNative --smoke-png)
    set_tests_properties(png_smoke PROPERTIES TIMEOUT 12 ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
    if(NOT WIN32)
        set_tests_properties(app_smoke PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
    endif()
endif()

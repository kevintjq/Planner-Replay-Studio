cmake_minimum_required(VERSION 3.22)
project(UAVStudio VERSION 0.1.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)

find_package(Qt6 6.8 REQUIRED COMPONENTS Quick QuickControls2 OpenGL Concurrent Multimedia Network Svg)
add_executable(UAVStudioAssets tools/render_assets.cpp)
target_link_libraries(UAVStudioAssets PRIVATE Qt6::Svg Qt6::Gui)
find_package(BZip2 REQUIRED)
find_package(OpenCV 4.8 REQUIRED COMPONENTS core imgproc photo tracking videoio)
find_package(yaml-cpp CONFIG REQUIRED)
find_package(lz4 CONFIG QUIET)
if(TARGET lz4::lz4)
    set(LZ4_LIBRARY lz4::lz4)
elseif(TARGET LZ4::lz4_shared)
    set(LZ4_LIBRARY LZ4::lz4_shared)
else()
    find_path(LZ4_INCLUDE_DIR NAMES lz4frame.h REQUIRED)
    find_library(LZ4_LIBRARY NAMES lz4 REQUIRED)
endif()

add_library(replay_data STATIC src/replay_data.cpp src/project_document.cpp src/video_timing.cpp src/rviz_layout.cpp src/transform_tree.cpp src/ros_topic_message.cpp)
target_include_directories(replay_data PUBLIC src)
if(LZ4_INCLUDE_DIR)
    target_include_directories(replay_data PRIVATE "${LZ4_INCLUDE_DIR}")
endif()
target_link_libraries(replay_data PUBLIC Qt6::Gui PRIVATE BZip2::BZip2 ${LZ4_LIBRARY} yaml-cpp::yaml-cpp)
add_library(trajectory_image STATIC src/trajectory_image.cpp src/trajectory_edits.cpp src/video_reader.cpp)
target_include_directories(trajectory_image PUBLIC src)
target_link_libraries(trajectory_image PUBLIC Qt6::Gui PRIVATE ${OpenCV_LIBS})
add_library(studio_appearance STATIC src/appearance_manager.cpp src/appearance_manager.h)
add_library(still_image STATIC src/still_image_writer.cpp src/frame_export_plan.cpp)
target_include_directories(still_image PUBLIC src)
target_link_libraries(still_image PUBLIC Qt6::Gui Qt6::Svg)
target_link_libraries(studio_appearance PUBLIC Qt6::Gui)
target_include_directories(studio_appearance PUBLIC src)

qt_add_executable(UAVStudio WIN32 src/main.cpp src/cloud_view.cpp src/cloud_view.h
    src/replay_controller.cpp src/replay_controller.h
    src/trajectory_controller.cpp src/trajectory_controller.h src/movie_exporter.cpp src/movie_exporter.h src/topic_inspector.cpp src/topic_inspector.h)
target_sources(UAVStudio PRIVATE src/trajectory_frame_model.cpp src/trajectory_frame_model.h src/appearance_acceptance.cpp src/frame_acceptance.cpp src/frame_batch_exporter.cpp src/frame_batch_exporter.h)
qt_add_resources(UAVStudio "interface" PREFIX "/" FILES qml/Main.qml qml/FlightClip.qml qml/VideoTimeline.qml
    qml/SceneOverlays.qml qml/SceneLayerEditor.qml qml/MarkerTimeline.qml qml/LayoutEditor.qml qml/TrajectoryPane.qml qml/TopicInspector.qml qml/SpeedGauge.qml
    qml/CloudInsets.qml qml/CloudInsetEditor.qml qml/LayerMotionEditor.qml qml/ReplayCanvas.qml qml/BatchExport.qml qml/TrajectoryTimeline.qml
    qml/StudioButton.qml qml/StudioComboBox.qml qml/StudioTextField.qml qml/StudioScrollBar.qml qml/StudioToolTip.qml qml/StudioCheckBox.qml qml/StudioSlider.qml qml/FrameExportPanel.qml qml/StudioLabel.qml qml/StudioPalette.qml qml/StudioSearchComboBox.qml)
target_link_libraries(UAVStudio PRIVATE replay_data trajectory_image studio_appearance still_image ${OpenCV_LIBS} Qt6::Quick Qt6::QuickControls2 Qt6::OpenGL Qt6::Concurrent Qt6::Multimedia)
if(MSVC)
    target_compile_options(replay_data PRIVATE /utf-8)
    target_compile_options(trajectory_image PRIVATE /utf-8)
    target_compile_options(still_image PRIVATE /utf-8)
    target_compile_options(studio_appearance PRIVATE /utf-8)
    target_compile_options(UAVStudio PRIVATE /utf-8)
endif()
add_executable(UAVStudioCli src/cli.cpp src/live_capture.cpp)
target_link_libraries(UAVStudioCli PRIVATE replay_data Qt6::Network)
if(MSVC)
    target_compile_options(UAVStudioCli PRIVATE /utf-8)
endif()
set_target_properties(UAVStudio PROPERTIES OUTPUT_NAME "uav-studio")
set_target_properties(UAVStudioCli PROPERTIES OUTPUT_NAME "uav-studio-cli")
set_source_files_properties(../assets/icons/uav-studio.svg PROPERTIES QT_RESOURCE_ALIAS "uav-studio.svg")
qt_add_resources(UAVStudio "branding" PREFIX "/brand" FILES ../assets/icons/uav-studio.svg)
target_link_libraries(UAVStudio PRIVATE Qt6::Svg)
if(WIN32)
    target_sources(UAVStudio PRIVATE resources/uav-studio.rc)
    install(TARGETS UAVStudio UAVStudioCli RUNTIME DESTINATION .)
elseif(APPLE)
    set_target_properties(UAVStudio PROPERTIES MACOSX_BUNDLE TRUE MACOSX_BUNDLE_BUNDLE_NAME "UAV Studio" MACOSX_BUNDLE_GUI_IDENTIFIER "org.uavstudio.app" MACOSX_BUNDLE_ICON_FILE "uav-studio.icns")
    target_sources(UAVStudio PRIVATE ../assets/icons/uav-studio.icns)
    set_source_files_properties(../assets/icons/uav-studio.icns PROPERTIES MACOSX_PACKAGE_LOCATION Resources)
    install(TARGETS UAVStudio BUNDLE DESTINATION .)
    install(TARGETS UAVStudioCli RUNTIME DESTINATION bin)
else()
    include(GNUInstallDirs)
    install(TARGETS UAVStudio UAVStudioCli RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
    install(FILES resources/uav-studio.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
    install(FILES resources/uav-studio-mime.xml DESTINATION ${CMAKE_INSTALL_DATADIR}/mime/packages)
    install(FILES ../assets/icons/uav-studio.svg DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps)
    foreach(size 16 32 48 64 128 256 512 1024)
        install(FILES ../assets/icons/uav-studio-${size}.png DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/${size}x${size}/apps RENAME uav-studio.png)
    endforeach()
endif()

cmake_minimum_required(VERSION 3.16)
project(UAVStudioRvizHost VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(catkin REQUIRED COMPONENTS rviz rosbag topic_tools tf2_msgs visualization_msgs rosgraph_msgs)
find_package(Qt5 5.12 REQUIRED COMPONENTS Widgets)
find_package(OpenGL REQUIRED)
find_package(yaml-cpp REQUIRED)
add_executable(uav-studio-rviz-host main.cpp bag_replay.cpp)
target_include_directories(uav-studio-rviz-host PRIVATE ${catkin_INCLUDE_DIRS})
target_link_libraries(uav-studio-rviz-host PRIVATE ${catkin_LIBRARIES} Qt5::Widgets OpenGL::GL yaml-cpp)
install(TARGETS uav-studio-rviz-host RUNTIME DESTINATION bin)

include(CTest)
if(BUILD_TESTING)
    add_executable(header-layout-test tests/header_layout_test.cpp)
    target_include_directories(header-layout-test PRIVATE ${catkin_INCLUDE_DIRS})
    target_link_libraries(header-layout-test PRIVATE Qt5::Core)
    add_test(NAME native-header-layout COMMAND header-layout-test)
endif()
